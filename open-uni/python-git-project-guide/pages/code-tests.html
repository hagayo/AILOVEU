src/github_client.py

"""Simple HTTP client examples using requests and httpx."""

from typing import Any

import httpx
import requests

API_URL = "https://api.github.com"


def get_github_status_with_requests() -> int:
    """Return GitHub API status code using requests."""
    response = requests.get(API_URL, timeout=10)
    return response.status_code


def get_github_info_with_httpx() -> dict[str, Any]:
    """Return basic GitHub API information using httpx."""
    response = httpx.get(API_URL, timeout=10)
    response.raise_for_status()
    return response.json()


def main() -> None:
    """Run a small demo."""
    status_code = get_github_status_with_requests()
    github_info = get_github_info_with_httpx()
    print(f"GitHub status code: {status_code}")
    print(f"Current user URL: {github_info.get("current_user_url")}")


if __name__ == "__main__":
    main()

tests/test_github_client.py

"""Tests for github_client module."""

from unittest.mock import Mock, patch

from src.github_client import get_github_info_with_httpx, get_github_status_with_requests


@patch("src.github_client.requests.get")
def test_get_github_status_with_requests(mock_get: Mock) -> None:
    """Test that requests client returns status code."""
    mock_response = Mock()
    mock_response.status_code = 200
    mock_get.return_value = mock_response

    result = get_github_status_with_requests()

    assert result == 200
    mock_get.assert_called_once()


@patch("src.github_client.httpx.get")
def test_get_github_info_with_httpx(mock_get: Mock) -> None:
    """Test that httpx client returns JSON data."""
    mock_response = Mock()
    mock_response.json.return_value = {"current_user_url": "https://api.github.com/user"}
    mock_response.raise_for_status.return_value = None
    mock_get.return_value = mock_response

    result = get_github_info_with_httpx()

    assert result["current_user_url"] == "https://api.github.com/user"
    mock_response.raise_for_status.assert_called_once()
