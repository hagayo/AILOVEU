tests/test_github_report.py

"""Tests for github_report module."""

import json

from src.github_report import save_report


def test_save_report_creates_json_file(tmp_path) -> None:
    """Test that save_report creates a JSON file."""
    output_file = tmp_path / "report.json"
    data = {"status_code": 200, "current_user_url": "https://api.github.com/user"}

    result_path = save_report(data, str(output_file))

    assert result_path == output_file
    assert output_file.exists()
    assert json.loads(output_file.read_text(encoding="utf-8")) == data

מוסיפים בדיקה לפונקציה החדשה build_github_summary:

from unittest.mock import patch

from src.github_client import build_github_summary


@patch("src.github_client.get_github_status_with_requests")
@patch("src.github_client.get_github_info_with_httpx")
def test_build_github_summary(mock_info, mock_status) -> None:
    """Test that build_github_summary returns selected fields."""
    mock_status.return_value = 200
    mock_info.return_value = {
        "current_user_url": "https://api.github.com/user",
        "authorizations_url": "https://api.github.com/authorizations",
        "repository_search_url": "https://api.github.com/search/repositories?q={query}",
    }

    result = build_github_summary()

    assert result["status_code"] == 200
    assert result["current_user_url"] == "https://api.github.com/user"
