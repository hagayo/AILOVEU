נחליף את הקובץ הקודם בגרסה מעט מקצועית יותר. הפונקציה מחזירה מידע מסודר ולא מערבבת הדפסות עם לוגיקה.

"""GitHub API client functions."""

from typing import Any

import httpx
import requests

API_URL = "https://api.github.com"


def get_github_status_with_requests() -> int:
    """Return GitHub API status code using requests."""
    response = requests.get(API_URL, timeout=10)
    return response.status_code


def get_github_info_with_httpx() -> dict[str, Any]:
    """Return basic GitHub API information using httpx."""
    response = httpx.get(API_URL, timeout=10)
    response.raise_for_status()
    return response.json()


def build_github_summary() -> dict[str, Any]:
    """Build a small summary from the GitHub API."""
    github_info = get_github_info_with_httpx()

    return {
        "status_code": get_github_status_with_requests(),
        "current_user_url": github_info.get("current_user_url"),
        "authorizations_url": github_info.get("authorizations_url"),
        "repository_search_url": github_info.get("repository_search_url"),
    }

למה זה יותר טוב?

  • הקוד שמדבר עם GitHub נמצא במקום אחד.
  • הפונקציות מחזירות מידע במקום להדפיס אותו מיד.
  • קל יותר לבדוק את הקוד עם Mock.
  • בעתיד אפשר להחליף API בלי לשבור את כל האפליקציה.
