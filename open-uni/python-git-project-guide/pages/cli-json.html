src/github_report.py

הקובץ הזה אחראי על שמירת הדוח לקובץ JSON. כך מפרידים בין קבלת מידע לבין שמירת מידע.

"""Create and save GitHub API reports."""

import json
from pathlib import Path
from typing import Any


def save_report(data: dict[str, Any], output_path: str) -> Path:
    """Save report data as a formatted JSON file."""
    path = Path(output_path)
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, indent=2), encoding="utf-8")
    return path

src/app.py

זהו קובץ ההפעלה הראשי של האפליקציה. הוא מקבל נתיב פלט דרך שורת הפקודה.

"""Command line entry point for the GitHub report app."""

import argparse

from src.github_client import build_github_summary
from src.github_report import save_report


def parse_args() -> argparse.Namespace:
    """Parse command line arguments."""
    parser = argparse.ArgumentParser(description="Create a small GitHub API report.")
    parser.add_argument(
        "--output",
        default="reports/github-summary.json",
        help="Path for the output JSON report.",
    )
    return parser.parse_args()


def main() -> None:
    """Run the application."""
    args = parse_args()
    summary = build_github_summary()
    output_path = save_report(summary, args.output)
    print(f"Report saved to: {output_path}")


if __name__ == "__main__":
    main()
